#include <iostream>
using namespace std;

int main() {
    float d1, d2;
    double D1, D2;

    cout << "Введіть діагоналі для float: ";
    cin >> d1 >> d2;
    cout << "Площа (float) = " << (d1 * d2) / 2 << endl;

    cout << "Введіть діагоналі для double: ";
    cin >> D1 >> D2;
    cout << "Площа (double) = " << (D1 * D2) / 2 << endl;

    return 0;
}
