#include <iostream>
#include <cmath>

using namespace std;

int main() {
    double a_d;
    double b_d;
    
    cout << "Введіть a_d: ";
    cin >> a_d;
    
    cout << "Введіть b_d: ";
    cin >> b_d;

    double num_double = pow(a_d + b_d, 4) - (pow(a_d, 4) + 4 * pow(a_d, 3) * b_d + 6 * pow(a_d, 2));
    double den_double = 4 * a_d * pow(b_d, 3) + pow(b_d, 4);
    double result_double = num_double / den_double;

    float a_f;
    float b_f;
    
    cout << "Введіть a_f: ";
    cin >> a_f;
    
    cout << "Введіть b_f: ";
    cin >> b_f;
    
    float num_float = powf(a_f + b_f, 4) - (powf(a_f, 4) + 4 * powf(a_f, 3) * b_f + 6 * powf(a_f, 2));
    float den_float = 4 * a_f * powf(b_f, 3) + powf(b_f, 4);
    float result_float = num_float / den_float;

    cout.precision(10);
    cout << "Результат типу float:  " << result_float << endl;
    cout << "Рузультат типу double: " << result_double << endl;

    return 0;
}
